// Engine: BPJS Claim Dispute Prevention Rule Matrix
export function validateClaimPreSubmission(claim: ClaimRecord): ValidationResult {
  const is144NonSpecialist = NonSpecialistList.has(claim.icd10);
  const isPolypharmacy = claim.prescriptions.length > 5;
  const isSepExpired = daysBetween(claim.sepDate, claim.serviceDate) > 15;

  return {
    allowSubmission: !is144NonSpecialist && !isSepExpired,
    disputeRisk: isPolypharmacy ? "HIGH_DRUG_AUDIT" : "CLEAN",
    ruleTags: ["BPJS-VCLAIM-2026", "PMK-NON-SPESIALISTIK"],
  };
}

export async function dispatchMcpTool(request: McpRequest): Promise<McpResponse> {
  const router = new NineRouterGateway({ endpoint: "127.0.0.1:20128" });
  const toolResult = await router.executeIsolated({
    tool: request.toolName,
